function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (data.action === 'saveNote') { return saveNoteHandler(ss, data.month, data.text); }
  if (data.action === 'update') {
    const ws = ss.getSheets()[0]; const row = parseInt(data.row) + 2;
    ws.getRange(row,1).setValue(data.name); ws.getRange(row,2).setValue(data.owner);
    ws.getRange(row,3).setValue(data.status); ws.getRange(row,5).setValue(data.startDate);
    ws.getRange(row,6).setValue(data.dueDate); ws.getRange(row,7).setValue(data.note);
    ws.getRange(row,8).setValue(data.priority); ws.getRange(row,9).setValue(data.tags);
    ws.getRange(row,10).setValue(data.parent); ws.getRange(row,11).setValue(data.hours);
    ws.getRange(row,12).setValue(data.comment);
    return ContentService.createTextOutput(JSON.stringify({result:'ok'})).setMimeType(ContentService.MimeType.JSON);
  }
  if (data.action === 'updateSort') {
    ss.getSheets()[0].getRange(parseInt(data.row)+2, 13).setValue(data.sort);
    return ContentService.createTextOutput(JSON.stringify({result:'ok'})).setMimeType(ContentService.MimeType.JSON);
  }
  if (data.action === 'delete') {
    ss.getSheets()[0].deleteRow(parseInt(data.row)+2);
    return ContentService.createTextOutput(JSON.stringify({result:'ok'})).setMimeType(ContentService.MimeType.JSON);
  }
  if (data.action === 'add' || !data.action) {
    const ws = ss.getSheets()[0];
    const maxSort = Math.max(0, ...ws.getRange(2,13,Math.max(ws.getLastRow()-1,1),1).getValues().flat().map(Number).filter(n=>!isNaN(n)));
    ws.appendRow([data.name,data.owner,data.status,'',data.startDate,data.dueDate,data.note,data.priority,data.tags,data.parent,data.hours,data.comment,maxSort+1]);
    return ContentService.createTextOutput(JSON.stringify({result:'ok'})).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({result:'unknown'})).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  if (e.parameter.action === 'saveNote') {
    return saveNoteHandler(SpreadsheetApp.getActiveSpreadsheet(), e.parameter.month, e.parameter.text);
  }
  return ContentService.createTextOutput(JSON.stringify({result:'unknown'})).setMimeType(ContentService.MimeType.JSON);
}
function saveNoteHandler(ss, month, text) {
  let ws = ss.getSheetByName('notes');
  if (!ws) { ws = ss.insertSheet('notes'); }
  const rows = ws.getDataRange().getValues();
  let found = false;
  for (let i = rows.length - 1; i >= 0; i--) {
    if (rows[i][0] === month) {
      if (!found) { ws.getRange(i+1, 2).setValue(text); found = true; }
      else { ws.deleteRow(i+1); }
    }
  }
  if (!found) ws.appendRow([month, text]);
  return ContentService.createTextOutput(JSON.stringify({result:'ok'})).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (data.action === 'clear') {
    const ws = ss.getSheetByName(data.month);
    if (ws && ws.getLastRow() > 1) ws.getRange(2, 1, ws.getLastRow()-1, 7).clearContent();
    return ContentService.createTextOutput(JSON.stringify({result:'ok'})).setMimeType(ContentService.MimeType.JSON);
  }
  if (data.action === 'add') {
    let ws = ss.getSheetByName(data.month);
    if (!ws) { ws = ss.insertSheet(data.month); ws.getRange(1,1,1,7).setValues([['負責人','工作項目','狀態','開始日','截止日','備註','工時']]); }
    ws.appendRow([data.owner, data.task, data.status, data.startDate, data.dueDate, data.note, data.hours]);
    return ContentService.createTextOutput(JSON.stringify({result:'ok'})).setMimeType(ContentService.MimeType.JSON);
  }
  if (data.action === 'saveZone') {
    let ws = ss.getSheetByName('zones');
    if (!ws) { ws = ss.insertSheet('zones'); ws.getRange(1,1,1,3).setValues([['負責人','區域','排序']]); }
    const rows = ws.getDataRange().getValues();
    let found = false;
    for (let i = 1; i < rows.length; i++) {
      if (rows[i][0] === data.owner) { ws.getRange(i+1,2).setValue(data.zone); ws.getRange(i+1,3).setValue(data.sort); found = true; break; }
    }
    if (!found) ws.appendRow([data.owner, data.zone, data.sort]);
    return ContentService.createTextOutput(JSON.stringify({result:'ok'})).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({result:'unknown'})).setMimeType(ContentService.MimeType.JSON);
}
